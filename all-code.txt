=============== src/components/QrCode.tsx ===============
const SIZE = 21

function hashString(value: string): number {
  let hash = 2166136261
  for (let index = 0; index < value.length; index += 1) {
    hash ^= value.charCodeAt(index)
    hash = Math.imul(hash, 16777619)
  }
  return hash >>> 0
}

function makeRandom(seed: number): () => number {
  let state = seed
  return () => {
    state |= 0
    state = (state + 0x6d2b79f5) | 0
    let t = Math.imul(state ^ (state >>> 15), 1 | state)
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

function isFinderCell(row: number, column: number): boolean {
  const origins = [
    [0, 0],
    [0, SIZE - 7],
    [SIZE - 7, 0],
  ]

  return origins.some(([top, left]) => {
    const localRow = row - top
    const localColumn = column - left
    if (localRow < 0 || localRow > 6 || localColumn < 0 || localColumn > 6) return false
    const border =
      localRow === 0 || localRow === 6 || localColumn === 0 || localColumn === 6
    const core = localRow >= 2 && localRow <= 4 && localColumn >= 2 && localColumn <= 4
    return border || core
  })
}

function isReserved(row: number, column: number): boolean {
  return (
    (row < 8 && column < 8) ||
    (row < 8 && column >= SIZE - 8) ||
    (row >= SIZE - 8 && column < 8)
  )
}

function buildMatrix(value: string): boolean[] {
  const random = makeRandom(hashString(value))
  const cells: boolean[] = []

  for (let row = 0; row < SIZE; row += 1) {
    for (let column = 0; column < SIZE; column += 1) {
      if (isFinderCell(row, column)) {
        cells.push(true)
      } else if (isReserved(row, column)) {
        cells.push(false)
      } else {
        cells.push(random() > 0.5)
      }
    }
  }

  return cells
}

interface QrCodeProps {
  value: string
  size?: number
}

export function QrCode({ value, size = 170 }: QrCodeProps) {
  const cells = buildMatrix(value)
  return (
    <div
      className="qr"
      role="img"
      aria-label={`QR code for ${value}`}
      style={{ width: size }}
    >
      {cells.map((filled, index) => (
        <i key={index} className={filled ? 'k' : ''} />
      ))}
    </div>
  )
}

=============== src/components/Reveal.tsx ===============
import { useEffect, useRef, type ReactNode, type CSSProperties } from 'react'

interface RevealProps {
  children: ReactNode
  delay?: number
  className?: string
}

export default function Reveal({ children, delay = 0, className = '' }: RevealProps) {
  const ref = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const el = ref.current
    if (!el) return
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          el.classList.add('is-visible')
          observer.disconnect()
        }
      },
      { threshold: 0.12 },
    )
    observer.observe(el)
    return () => observer.disconnect()
  }, [])

  const style = { '--d': `${delay}ms` } as CSSProperties

  return (
    <div ref={ref} style={style} className={`reveal ${className}`}>
      {children}
    </div>
  )
}

import { Link } from '@tanstack/react-router'
import type { NavLink } from '../routes/links'

interface SiteLinkProps {
  item: NavLink
  className?: string
  onClick?: () => void
}

export default function SiteLink({ item, className, onClick }: SiteLinkProps) {
  if (item.slug) {
    return (
      <a href={`#${item.slug}`} className={className} onClick={onClick}>
        {item.label}
      </a>
    )
  }
  if (item.to) {
    return (
      <Link to={item.to} className={className} onClick={onClick}>
        {item.label}
      </Link>
    )
  }
  return (
    <a href={item.href} className={className} onClick={onClick}>
      {item.label}
    </a>
  )
}

=============== src/features/authentication/AuthProvider.tsx ===============
import { useState, type ReactNode } from "react";
import { AuthContext } from "./useAuth";
import { getUser,loginRequest,logoutRequest,registerRequest,type User } from "./authStore";


export function AuthProvider({children}:{children: ReactNode}){
    const [user,setUser] = useState<User | null>(getUser);

    const login = async (email:string,password:string)=>{
        const next = await loginRequest(email,password);
        setUser(next);
        return next;
    };
    const register = async (name:string,email:string,password:string) => {
        const next = await registerRequest(name,email,password);
        setUser(next);
        return next;
    };
    const logout = () => {
        logoutRequest();
        setUser(null);
    };
    return <AuthContext.Provider value={{user,login,register,logout}}>{children}</AuthContext.Provider>;
}

const SESSION_KEY = 'tm_user';

function readJson<T>(key:string): T | null {
    try{
        const raw = localStorage.getItem(key);
        return raw ? (JSON.parse(raw) as T   ) : null;
    }catch{
        return null;
    }
}
function allAccounts():StoredUser[]{
    return [...MOCK_USERS, ...(readJson<StoredUser[]>(ACCOUNTS_KEY)?? [])];
}
const wait = ()=> new Promise((resolve) => setTimeout(resolve,400));

export function getUser(): User | null{
    return readJson<User>(SESSION_KEY);
}

function startSession(account : StoredUser): User {
    const user:User = {id:account.id,name:account.name,email:account.email,role:account.role};
    localStorage.setItem(SESSION_KEY,JSON.stringify(user));
    return user;
}
export const DEMO_ACCOUNTS = [

] as const

export async function loginRequest(email:string,password:string):Promise<User>{
    await wait();
    const account = allAccounts().find((a) => a.email ===email.trim().toLowerCase() && a.password ===password);
    if(!account) throw new Error('Wrong email or password');
    return startSession(account);
}

export async function registerRequest(name:string,email:string,password:string): Promise<User>{
    await wait();
    const cleanEmail = email.trim().toLowerCase();
    if(allAccounts().some((a) => a.email===cleanEmail)) throw new Error('This account already exists');
    const account: StoredUser = {id: `u-${Date.now()}`,name:name.trim(),email:cleanEmail,password,role:'customer'};
    const saved = readJson<StoredUser[]>(ACCOUNTS_KEY)??[];
    localStorage.setItem(ACCOUNTS_KEY,JSON.stringify([...saved,account]));
    return startSession(account);
}

export function logoutRequest():void{
    localStorage.removeItem(SESSION_KEY);
}

import { useState, type FormEvent } from 'react';
import { useMutation } from '@tanstack/react-query';
import { useNavigate, useSearch } from '@tanstack/react-router';
import type { User } from '../authStore';
import { DEMO_ACCOUNTS } from '../authStore';
import { useAuth } from '../useAuth';
import styles from './LoginPage.module.css';

function homeForRole(role: User['role']): string {
  if (role === 'admin') return '/admin';        // <-- Added Admin redirect
  if (role === 'organizer') return '/organizer/dashboard';
  if (role === 'staff') return '/scan';
  return '/events';
}

type Mode = 'login' | 'register';

function validate(mode: Mode, name: string, email: string, password: string): string | null {
  if (mode === 'register' && !name.trim()) return 'Enter your full name';
  if (!/^\S+@\S+\.\S+$/.test(email)) return 'Enter a valid email address';
  if (password.length < 8) return 'Password must be at least 8 characters';
  return null;
}

// Only allow redirects to pages inside this app
function safePath(path: string | undefined): string {
  return path && path.startsWith('/') && !path.startsWith('//') ? path : '/events';
}

export function LoginPage() {
  const { login, register } = useAuth();
  const navigate = useNavigate();
  const { redirect } = useSearch({ strict: false }) as { redirect?: string };

  const [mode, setMode] = useState<Mode>('login');
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);
  const [formError, setFormError] = useState<string | null>(null);

  const goAfterLogin = (user: User) =>
    navigate({ to: redirect ? safePath(redirect) : homeForRole(user.role) });

  const submit = useMutation({
    mutationFn: () => (mode === 'login' ? login(email, password) : register(name, email, password)),
    onSuccess: goAfterLogin,
  });

  const demoLogin = useMutation({
    mutationFn: (account: (typeof DEMO_ACCOUNTS)[number]) =>
      login(account.email, account.password),
    onSuccess: goAfterLogin,
  });

  const onSubmit = (e: FormEvent) => {
    e.preventDefault();
    const error = validate(mode, name, email, password);
    setFormError(error);
    if (!error) submit.mutate();
  };

  const switchMode = (next: Mode) => {
    setMode(next);
    setFormError(null);
    submit.reset();
  };

  const errorMessage =
    formError ?? (submit.error?.message || demoLogin.error?.message || null);

  return (
    <main className={styles.page}>
      <aside className={styles.brand}>
        <div className={styles.logo}>
          <span className={styles.logoMark}></span>
          Ticket Master
        </div>

        <div>
          <h1 className={styles.headline}>Your next great night starts here.</h1>
          <p className={styles.tagline}>Discover events, buy tickets in minutes and walk in with a simple scan.</p>
          <ul className={styles.perks}>
            <li><span className={styles.perkIcon}>✓</span>Find events that match your taste</li>
            <li><span className={styles.perkIcon}>✓</span>Pay once and get your ticket instantly</li>
            <li><span className={styles.perkIcon}>✓</span>Show your QR code at the door</li>
          </ul>
        </div>
      </aside>

      <section className={styles.formSide}>
        <div className={styles.card}>
          <h2 className={styles.title}>{mode === 'login' ? 'Welcome back' : 'Create your account'}</h2>
          <p className={styles.subtitle}>
            {mode === 'login' ? 'Log in to buy tickets and see your bookings.' : 'It only takes a minute to get started.'}
          </p>

          <div className={styles.tabs}>
            <button type="button" onClick={() => switchMode('login')} className={`${styles.tab} ${mode === 'login' ? styles.active : ''}`}>
              Log in
            </button>
            <button type="button" onClick={() => switchMode('register')} className={`${styles.tab} ${mode === 'register' ? styles.active : ''}`}>
              Register
            </button>
          </div>

          <form onSubmit={onSubmit} noValidate>
            {mode === 'register' && (
              <input placeholder="Full name" value={name} onChange={(e) => setName(e.target.value)} className={styles.field} />
            )}
            <input type="email" placeholder="Email address" value={email} onChange={(e) => setEmail(e.target.value)} className={styles.field} />
            <div className={styles.passwordRow}>
              <input
                type={showPassword ? 'text' : 'password'}
                placeholder="Password"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                className={styles.field}
              />
              <button type="button" onClick={() => setShowPassword((s) => !s)} className={styles.toggle}>
                {showPassword ? 'Hide' : 'Show'}
              </button>
            </div>

            {errorMessage && <p role="alert" className={styles.error}>{errorMessage}</p>}

            <button type="submit" disabled={submit.isPending} className={styles.submit}>
              {submit.isPending ? 'Please wait…' : mode === 'login' ? 'Log in' : 'Create account'}
            </button>
          </form>

          <p className={styles.note}>Browsing events never needs an account.</p>

          {mode === 'login' && (
            <div className={styles.demo}>
              <p className={styles.demoTitle}>Demo accounts · one-tap sign in</p>
              <div className={styles.demoButtons}>
                {DEMO_ACCOUNTS.map((account) => (
                  <button
                    key={account.email}
                    type="button"
                    disabled={demoLogin.isPending || submit.isPending}
                    onClick={() => demoLogin.mutate(account)}
                    className={styles.demoButton}
                  >
                    {account.role}
                  </button>
                ))}
              </div>
            </div>
          )}
        </div>
      </section>
    </main>
  );
}

      </button>

  quantity: number
  onChangeQuantity: (quantity: number) => void
}

export function TicketSummaryCard({
  event,
  ticketType,
  quantity,
  onChangeQuantity,
}: TicketSummaryCardProps): ReactNode {
  const remaining = ticketType.capacity - ticketType.sold
  const soldOut = remaining <= 0

  return (
    <div className="card row">
      <div>
        <b>{event.name}</b>
        <div className="m">
          {event.date} · {ticketType.name} · {event.venue}, {event.city}
        </div>
        <div className="m">
          {formatKwacha(ticketType.price)} each · {soldOut ? 'Sold out' : `${remaining} left`}
        </div>
      </div>
      <QuantityStepper
        value={quantity}
        min={1}
        max={Math.max(remaining, 1)}
        onChange={onChangeQuantity}
      />
    </div>
  )
}

  quantity: number,

import { useMutation } from '@tanstack/react-query'
import { issueBooking } from '../../../lib/tickets'
import { getEvent, incrementSold } from '../../../lib/events'
import type { CheckoutInput, CheckoutResult, PaymentDetails } from '../types'

export class CheckoutError extends Error {}

function delay(ms: number): Promise<void> {
  return new Promise((resolve) => {
    setTimeout(resolve, ms)
  })
}

export function validatePayment(payment: PaymentDetails): string | null {
  if (payment.method === 'mobile_money') {
    if (!payment.provider) return 'Choose a mobile money provider.'
    const phone = payment.phoneNumber.replace(/\s+/g, '')
    if (!/^\d{9,12}$/.test(phone)) return 'Enter a valid mobile money phone number.'
    return null
  }

  const card = payment.cardNumber.replace(/\s+/g, '')
  if (!/^\d{16}$/.test(card)) return 'Enter a valid 16-digit card number.'
  if (!/^\d{2}\/\d{2}$/.test(payment.cardExpiry)) return 'Enter the card expiry as MM/YY.'
  return null
}

function isPaymentAccepted(payment: PaymentDetails): boolean {
  const reference =
    payment.method === 'mobile_money' ? payment.phoneNumber : payment.cardNumber
  const digits = reference.replace(/\s+/g, '')
  return !digits.endsWith('00')
}

async function submitCheckout(input: CheckoutInput): Promise<CheckoutResult> {
  await delay(900)

  const remaining = input.ticketType.capacity - input.ticketType.sold
  if (input.quantity < 1) {
    throw new CheckoutError('Select at least one ticket.')
  }
  if (remaining <= 0) {
    throw new CheckoutError(`${input.ticketType.name} tickets are sold out for this event.`)
  }
  if (input.quantity > remaining) {
    throw new CheckoutError(
      `Only ${remaining} ${input.ticketType.name} ticket${remaining === 1 ? '' : 's'} left.`,
    )
  }

  const paymentError = validatePayment(input.payment)
  if (paymentError) throw new CheckoutError(paymentError)

  if (!isPaymentAccepted(input.payment)) {
    throw new CheckoutError('Payment was declined. No ticket was issued — please try again.')
  }

  const storedEvent = getEvent(input.event.id)
  if (!storedEvent || storedEvent.status !== 'Published') {
    throw new CheckoutError('This event is not available for booking.')
  }

  // One call creates the tickets AND the booking that owns them (Booking History reads it)
  const { tickets } = issueBooking({
    event: storedEvent,
    ticketType: input.ticketType,
    quantity: input.quantity,
    holder: input.holder,
  })

  // Persist the sale so remaining seats drop across the whole app.
  incrementSold(input.event.id, input.ticketType.id, input.quantity)

  return { reference: tickets[0].id, tickets }
}

export function useCheckout() {
  return useMutation({ mutationFn: submitCheckout })
}

export function CheckoutPage(): ReactNode {
  const { eventId } = useParams({ strict: false }) as { eventId?: string }
  const { type, qty } = useSearch({ strict: false }) as { type?: string; qty?: string }
  const parsedQty = parseInt(qty ?? '', 10)

  const [selectedTypeId, setSelectedTypeId] = useState<string | null>(null)
  const [quantity, setQuantity] = useState(() =>
    Number.isNaN(parsedQty) || parsedQty < 1 ? 1 : parsedQty,
  )
  const { user } = useAuth()
  const [holder, setHolder] = useState(user?.name ?? '')
  const [payment, setPayment] = useState<PaymentDetails>(EMPTY_PAYMENT)
  const [clientError, setClientError] = useState<string | null>(null)
  const [result, setResult] = useState<CheckoutResult | null>(null)

  const { data: eventData, isLoading, isError } = useQuery({
    queryKey: ['checkout-event', eventId],
    queryFn: () => {
      const found = getEvent(eventId ?? '')
      // drafts are private to the organizer, so they cannot be bought
      return withDelay(found && found.status === 'Published' ? found : null)
    },
    enabled: !!eventId,
  })

  const ticketTypes: TicketType[] = eventData ? toCheckoutTicketTypes(eventData) : []
  const activeTypeId = selectedTypeId ?? type ?? ticketTypes[0]?.id
  const ticketType =
    ticketTypes.find((candidate) => candidate.id === activeTypeId) ?? ticketTypes[0]
  const checkoutEvent: CheckoutEvent = eventData
    ? toCheckoutEvent(eventData)
    : {
        id: eventId ?? 'unknown',
        name: 'Event',
        date: '',
        time: '',
        venue: '',
        city: '',
        organizer: '',
      }

  const checkout = useCheckout()
  const queryClient = useQueryClient()

  const totals = ticketType
    ? calculateTotals(ticketType.price, quantity)
    : { subtotal: 0, serviceFee: 0, total: 0 }
  const remaining = ticketType ? remainingSeats(ticketType) : 0
  const soldOut = remaining <= 0
  const exceedsAvailable = quantity > remaining
  function updatePayment(patch: Partial<PaymentDetails>) {
    setPayment((current) => ({ ...current, ...patch }))
  }

  function selectTicketType(id: string) {
    setSelectedTypeId(id)
    setQuantity(1)
    setClientError(null)
  }

  function handlePay() {
    if (!ticketType) return
    setClientError(null)
    const paymentError = validatePayment(payment)
    if (paymentError) {
      setClientError(paymentError)
      return
    }
    checkout.mutate(
      { event: checkoutEvent, ticketType, quantity, holder, payment },
      {
        onSuccess: (data) => {
          setResult(data)
          queryClient.invalidateQueries({ queryKey: ['events'] })
          queryClient.invalidateQueries({ queryKey: ['checkout-event'] })
          queryClient.invalidateQueries({ queryKey: ['tickets'] })
          queryClient.invalidateQueries({ queryKey: ['bookings'] })
        },
      },
    )

          setQuantity(1)

  const errorMessage = clientError ?? mutationError

                    disabled={disabled}

=============== src/features/landing/components/SearchBar.tsx ===============
import { useEffect, useRef, useState, type ChangeEvent } from 'react'
import { Link } from '@tanstack/react-router'
import { Search, SlidersHorizontal } from 'lucide-react'
import { useEvents } from '../../events'
import {
  categories,
  cities,
  priceOptions,
  sortOptions,
  defaultFilters,
  filterEvents,
  type Filters,
} from '../filters'

const selectStyle =
  'mt-1 w-full rounded-xl border border-slate-200 bg-white px-3 py-2.5 text-sm font-normal text-slate-700 outline-none focus:border-navy'

interface HighlightProps {
  text: string
  word: string
}

function Highlight({ text, word }: HighlightProps) {
  const start = text.toLowerCase().indexOf(word.toLowerCase())
  if (!word || start === -1) return <>{text}</>
  const end = start + word.length
  return (
    <>
      {text.slice(0, start)}
      <b className="text-navy">{text.slice(start, end)}</b>
      {text.slice(end)}
    </>
  )
}

interface SearchBarProps {
  filters: Filters
  setFilters: (f: Filters) => void
}

type OpenBox = 'none' | 'filters' | 'results'

export default function SearchBar({ filters, setFilters }: SearchBarProps) {
  const [openBox, setOpenBox] = useState<OpenBox>('none')
  const wrapperRef = useRef<HTMLDivElement>(null)
  const { data: events = [] } = useEvents()

  useEffect(() => {
    const handleClick = (e: MouseEvent) => {
      const target = e.target as Node
      if (wrapperRef.current && !wrapperRef.current.contains(target)) setOpenBox('none')
    }
    document.addEventListener('mousedown', handleClick)
    return () => document.removeEventListener('mousedown', handleClick)
  }, [])

  const update = (name: keyof Filters, value: string) => setFilters({ ...filters, [name]: value })

  const handleTyping = (e: ChangeEvent<HTMLInputElement>) => {
    update('search', e.target.value)
    setOpenBox(e.target.value.trim() ? 'results' : 'none')
  }

  const activeCount = (['category', 'city', 'price'] as const).filter(
    (name) => filters[name] !== defaultFilters[name],
  ).length

  const matches = filterEvents(events, filters)
  const word = filters.search.trim()

  const goToEvents = () => {
    setOpenBox('none')
    document.getElementById('events')?.scrollIntoView({ behavior: 'smooth' })
  }

  return (
    <div ref={wrapperRef} className="relative mx-auto w-full max-w-3xl">
      <form
        onSubmit={(e) => {
          e.preventDefault()
          goToEvents()
        }}
        className="flex items-center gap-2 rounded-full bg-white p-2 pl-5 shadow-2xl"
      >
        <Search size={20} className="shrink-0 text-slate-400" />
        <input
          value={filters.search}
          onChange={handleTyping}
          placeholder="Search concerts, sports, workshops..."
          className="min-w-0 flex-1 bg-transparent py-2 text-sm text-slate-700 outline-none sm:text-base"
        />

        <button
          type="button"
          aria-label="Filters"
          onClick={() => setOpenBox(openBox === 'filters' ? 'none' : 'filters')}
          className={`relative flex h-11 w-11 shrink-0 items-center justify-center rounded-full ${
            openBox === 'filters' ? 'bg-navy text-white' : 'bg-navy-50 text-navy hover:bg-navy-100'
          }`}
        >
          <SlidersHorizontal size={20} />
          {activeCount > 0 && (
            <span className="absolute -right-1 -top-1 flex h-5 w-5 items-center justify-center rounded-full bg-navy-dark text-[11px] font-bold text-white">
              {activeCount}
            </span>
          )}
        </button>

        <button
          type="submit"
          className="hidden h-11 rounded-full bg-navy px-6 text-sm font-semibold text-white hover:bg-navy-dark sm:block"
        >
          Search
        </button>
      </form>

      {openBox === 'results' && (
        <div className="absolute left-0 right-0 top-full z-30 mt-3 max-h-[70vh] overflow-y-auto rounded-2xl bg-white p-2 text-left shadow-2xl">
          {matches.length === 0 && <p className="p-4 text-sm">No events found for "{word}"</p>}

          {matches.slice(0, 5).map((event) => (
            <Link
              key={event.id}
              to="/events/$eventId"
              params={{ eventId: String(event.id) }}
              onClick={() => setOpenBox('none')}
              className="flex items-center gap-3 rounded-xl p-2 hover:bg-navy-50"
            >
              <img src={event.image} alt="" className="h-12 w-12 rounded-lg object-cover" />
              <span>
                <span className="block font-semibold text-slate-700">
                  <Highlight text={event.title} word={word} />
                </span>
                <span className="text-xs">
                  {event.category} · {event.city}
                </span>
              </span>
            </Link>
          ))}

          {matches.length > 0 && (
            <button
              onClick={goToEvents}
              className="w-full rounded-xl p-3 text-sm font-semibold text-navy hover:bg-navy-50"
            >
              See all {matches.length} result(s)
            </button>
          )}
        </div>
      )}

      {openBox === 'filters' && (
        <div className="absolute left-0 right-0 top-full z-30 mt-3 max-h-[70vh] overflow-y-auto rounded-2xl bg-white p-5 text-left shadow-2xl">
          <p className="text-sm font-semibold text-navy-dark">Category</p>
          <div className="mt-2 flex flex-wrap gap-2">
            {categories.map((c) => (
              <button
                key={c}
                onClick={() => update('category', c)}
                className={`rounded-full px-4 py-1.5 text-sm font-medium ${
                  filters.category === c ? 'bg-navy text-white' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'
                }`}
              >
                {c}
              </button>
            ))}
          </div>

          <div className="mt-4 grid gap-3 sm:grid-cols-3">
            <label className="text-sm font-semibold text-navy-dark">
              City
              <select value={filters.city} onChange={(e) => update('city', e.target.value)} className={selectStyle}>
                {cities.map((c) => (
                  <option key={c}>{c}</option>
                ))}
              </select>
            </label>
            <label className="text-sm font-semibold text-navy-dark">
              Price
              <select value={filters.price} onChange={(e) => update('price', e.target.value)} className={selectStyle}>
                {priceOptions.map((p) => (

                    {p.label}

                ))}
              </select>
            </label>
            <label className="text-sm font-semibold text-navy-dark">
              Sort by
              <select value={filters.sort} onChange={(e) => update('sort', e.target.value)} className={selectStyle}>
                {sortOptions.map((s) => (
                  <option key={s.value} value={s.value}>
                    {s.label}
                  </option>
                ))}
              </select>
            </label>
          </div>

          <div className="mt-5 flex items-center justify-between">
            <button
              onClick={() => setFilters(defaultFilters)}
              className="text-sm font-semibold text-slate-500 hover:text-navy"
            >
              Reset all
            </button>
            <button
              onClick={goToEvents}
              className="rounded-full bg-navy px-6 py-2 text-sm font-semibold text-white hover:bg-navy-dark"
            >
              Show {matches.length} event(s)
            </button>
          </div>
        </div>
      )}
    </div>
  )
}

import { fromPrice, type Event } from '../../types/event'

export const categories = ['All', 'Concerts', 'Sports', 'Festivals', 'Workshops', 'Community']
export const cities = ['All cities', 'Lilongwe', 'Blantyre', 'Mzuzu', 'Zomba', 'Mangochi', 'Salima']

export const priceOptions = [
  { value: 'any', label: 'Any price' },
  { value: 'free', label: 'Free' },
  { value: 'under10', label: 'Under MK 10,000' },
  { value: 'under20', label: 'Under MK 20,000' },
  { value: 'over20', label: 'MK 20,000 and above' },
]

export const sortOptions = [
  { value: 'date', label: 'Soonest first' },
  { value: 'low', label: 'Price: low to high' },
  { value: 'high', label: 'Price: high to low' },
]

export interface Filters {
  search: string
  category: string
  city: string
  price: string
  sort: string
}

export const defaultFilters: Filters = {
  search: '',
  category: 'All',
  city: 'All cities',
  price: 'any',
  sort: 'date',
}

function matchesPrice(price: number, option: string): boolean {
  if (option === 'free') return price === 0
  if (option === 'under10') return price < 10000
  if (option === 'under20') return price < 20000
  if (option === 'over20') return price >= 20000
  return true
}

export function filterEvents(events: Event[], filters: Filters): Event[] {
  const search = filters.search.trim().toLowerCase()

  return events
    .filter((e) => filters.category === 'All' || e.category === filters.category)
    .filter((e) => filters.city === 'All cities' || e.city === filters.city)
    .filter((e) => matchesPrice(fromPrice(e), filters.price))
    .filter((e) =>
      `${e.title} ${e.city} ${e.venue} ${e.category}`.toLowerCase().includes(search),
    )
    .sort((a, b) => {
      if (filters.sort === 'low') return fromPrice(a) - fromPrice(b)
      if (filters.sort === 'high') return fromPrice(b) - fromPrice(a)
      return new Date(a.date).getTime() - new Date(b.date).getTime()
    })
}

  { key: 'confirmed', label: 'Confirmed' },
  { key: 'cancelled', label: 'Cancelled' },
];

export function BookingHistoryPage() {
  const { data, isLoading, isError, refetch } = useBookings();

    setSelectedId(id);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  }

    <div className={styles.page}>
      <header className={styles.header}>
        <h1 className={styles.heading}>My Tickets</h1>
        <p className={styles.sub}>Your tickets, QR codes and entry status</p>
      </header>

      {isError && (

          Could not load tickets.{' '}
          <button type="button" className={styles.retry} onClick={() => refetch()}>
            Try again
          </button>
        </p>
      )}

      {!isLoading && !isError && !featured && (
        <p className={styles.state}>You have no tickets yet.</p>
      )}

      {featured && <TicketDetail ticket={featured} />}

      {others.length > 0 && (
        <section className={styles.section}>
          <h2 className={styles.sectionTitle}>Other tickets</h2>
          <div className={styles.list}>
            {others.map((t) => (
              <article key={t.id} className={styles.item}>
                <div className={styles.itemInfo}>
                  <h3 className={styles.itemTitle}>{t.eventTitle}</h3>
                  <p className={styles.meta}>
                    {t.venue} · {formatDate(t.date)}
                  </p>
                </div>
                <div className={styles.itemSide}>
                  <span
                    className={`${styles.badge} ${t.status === 'valid' ? styles.valid : styles.used}`}
                  >
                    {t.status}
                  </span>
                  <button
                    type="button"
                    className={styles.retry}
                    onClick={() => handleView(t.id)}
                  >
                    View ticket
                  </button>
                </div>
              </article>
            ))}
          </div>
        </section>
      )}

      {tickets.length > 0 && (
        <section className={styles.section}>
          <h2 className={styles.sectionTitle}>All bookings</h2>

          <div className={styles.filters}>
            {tabs.map((tabItem) => (
              <button
                key={tabItem.key}
                type="button"
                className={`${styles.pill} ${tab === tabItem.key ? styles.pillActive : ''}`}
                onClick={() => setTab(tabItem.key)}
              >
                {tabItem.label}
              </button>
            ))}
          </div>

          {visible.length === 0 ? (
            <p className={styles.state}>No tickets to show.</p>
          ) : (
            <TicketsTable tickets={visible} onView={handleView} />
          )}
        </section>
      )}
    </div>
  );
}

import type { Ticket } from './ticketTypes';
import { formatDate, formatMoney } from './ticketFormat';

export function downloadTicketImage(
  ticket: Ticket,
  qrCanvas: HTMLCanvasElement | null,
): void {
  const canvas = document.createElement('canvas');
  canvas.width = 800;
  canvas.height = 380;
  const ctx = canvas.getContext('2d');
  if (!ctx) return;

  // background + header band
  ctx.fillStyle = '#ffffff';
  ctx.fillRect(0, 0, 800, 380);
  ctx.fillStyle = '#12326b';
  ctx.fillRect(0, 0, 800, 70);
  ctx.fillStyle = '#ffffff';
  ctx.font = 'bold 26px sans-serif';
  ctx.textBaseline = 'middle';
  ctx.fillText('TICKET MASTER', 32, 35);
  ctx.textBaseline = 'alphabetic';

  // title + venue
  ctx.fillStyle = '#0a1f47';
  ctx.font = 'bold 28px sans-serif';
  ctx.fillText(ticket.eventTitle, 32, 125, 500);
  ctx.fillStyle = '#64748b';
  ctx.font = '18px sans-serif';
  ctx.fillText(ticket.venue, 32, 155, 500);

  // details
  const lines: [string, string][] = [
    ['Date', formatDate(ticket.date)],
    ['Time', ticket.time],
    ['Ticket', ticket.ticketType],
    ['Entry', ticket.seat],
    ['Price', formatMoney(ticket.price)],
    ['Status', ticket.status],
  ];
  lines.forEach(([label, value], i) => {
    const y = 205 + i * 28;
    ctx.fillStyle = '#64748b';
    ctx.font = 'bold 16px sans-serif';
    ctx.fillText(label, 32, y);
    ctx.fillStyle = '#0a1f47';
    ctx.font = '16px sans-serif';
    ctx.fillText(value, 120, y, 380);
  });

  // QR code + ticket id
  if (qrCanvas) {
    ctx.drawImage(qrCanvas, 580, 110, 180, 180);
  }
  ctx.fillStyle = '#64748b';
  ctx.font = 'bold 15px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText(ticket.id, 670, 320);

  canvas.toBlob((blob) => {
    if (!blob) return;
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `${ticket.id}.png`;
    link.click();
    URL.revokeObjectURL(url);
  }, 'image/png');
}

export function formatDate(iso: string): string {
  if (!iso) return 'TBD';
  return new Date(`${iso}T00:00:00`).toLocaleDateString('en-GB', {
    day: 'numeric',
    month: 'short',
    year: 'numeric',
  });
}

export function formatMoney(amount: number | undefined | null): string {
  if (amount === undefined || amount === null || isNaN(amount)) return 'MK 0';
  return `MK ${amount.toLocaleString()}`;
}

// true when the event date is today or later
export function isUpcoming(iso: string): boolean {
  if (!iso) return false;
  const today = new Date().toLocaleDateString('en-CA'); // YYYY-MM-DD
  return iso >= today;
}

import { useCallback, useEffect, useRef, useState } from 'react'

interface DetectedBarcode {
  rawValue: string
}

interface BarcodeDetectorLike {
  detect: (source: CanvasImageSource) => Promise<DetectedBarcode[]>
}

interface BarcodeDetectorConstructor {
  new (options?: { formats?: string[] }): BarcodeDetectorLike
}

export type QrScannerStatus =
  | 'idle'
  | 'starting'
  | 'scanning'
  | 'unsupported'
  | 'denied'
  | 'error'

const SCAN_INTERVAL_MS = 400
const REPEAT_COOLDOWN_MS = 2500

function getBarcodeDetectorConstructor(): BarcodeDetectorConstructor | undefined {
  return (window as unknown as { BarcodeDetector?: BarcodeDetectorConstructor }).BarcodeDetector
}

export function useQrScanner(onDetected: (code: string) => void) {
  const videoRef = useRef<HTMLVideoElement>(null)
  const streamRef = useRef<MediaStream | null>(null)
  const detectorRef = useRef<BarcodeDetectorLike | null>(null)
  const timerRef = useRef<number | null>(null)
  const lastCodeRef = useRef<{ code: string; at: number }>({ code: '', at: 0 })
  const onDetectedRef = useRef(onDetected)

  const [status, setStatus] = useState<QrScannerStatus>('idle')
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    onDetectedRef.current = onDetected
  }, [onDetected])

  const stop = useCallback(() => {
    if (timerRef.current !== null) {
      window.clearInterval(timerRef.current)
      timerRef.current = null
    }
    if (streamRef.current) {
      streamRef.current.getTracks().forEach((track) => track.stop())
      streamRef.current = null
    }
    if (videoRef.current) {
      videoRef.current.srcObject = null
    }
    detectorRef.current = null
    setStatus('idle')
  }, [])

  const start = useCallback(async () => {
    setError(null)

    const Detector = getBarcodeDetectorConstructor()
    if (!Detector) {
      setStatus('unsupported')
      setError('QR scanning is not supported in this browser. Enter the ticket ID manually below.')
      return
    }
    if (!navigator.mediaDevices?.getUserMedia) {
      setStatus('unsupported')
      setError('Camera access is not available here. Enter the ticket ID manually below.')
      return
    }

    setStatus('starting')
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: 'environment' },
        audio: false,
      })
      streamRef.current = stream

      const video = videoRef.current
      if (!video) {
        stop()
        return
      }
      video.srcObject = stream
      await video.play()

      detectorRef.current = new Detector({ formats: ['qr_code'] })
      setStatus('scanning')

      timerRef.current = window.setInterval(async () => {
        const currentVideo = videoRef.current
        const detector = detectorRef.current
        if (!currentVideo || !detector || currentVideo.readyState < 2) return
        try {
          const codes = await detector.detect(currentVideo)
          const value = codes[0]?.rawValue?.trim()
          if (!value) return
          const now = Date.now()
          if (value === lastCodeRef.current.code && now - lastCodeRef.current.at < REPEAT_COOLDOWN_MS) {
            return
          }
          lastCodeRef.current = { code: value, at: now }
          onDetectedRef.current(value)
        } catch {
          // Ignore transient decoding errors and keep scanning.
        }
      }, SCAN_INTERVAL_MS)
    } catch (caught) {
      stop()
      const denied = caught instanceof DOMException && caught.name === 'NotAllowedError'
      setStatus(denied ? 'denied' : 'error')
      setError(
        denied
          ? 'Camera permission was denied. Enter the ticket ID manually below.'
          : 'Could not start the camera. Enter the ticket ID manually below.',
      )
    }
  }, [stop])

  useEffect(() => stop, [stop])

  return { videoRef, status, error, start, stop }
}

import {
  getTicket,
  markTicketUsed,
  normalizeTicketCode,
  type TicketValidationResult,
} from '../../../lib/tickets'

export function validateTicket(rawCode: string, eventId: string): TicketValidationResult {
  const code = normalizeTicketCode(rawCode)

  if (!code) {
    return {
      status: 'invalid',
      title: 'Invalid ticket',
      message: 'Enter or scan a ticket ID.',
    }
  }

  const ticket = getTicket(code)
  if (!ticket) {
    return {
      status: 'invalid',
      title: 'Invalid ticket',
      message: `“${code}” is not a ticket issued for this platform.`,
    }
  }

  if (ticket.eventId !== eventId) {
    return {
      status: 'wrong_event',
      title: 'Wrong event',
      message: `${ticket.id} is for ${ticket.eventName}, not this event.`,
      ticket,
    }
  }

  if (ticket.status === 'used') {
    return {
      status: 'used',
      title: 'Already used',
      message: `${ticket.id} has already been admitted and cannot be accepted again.`,
      ticket,
    }
  }

  const admitted = markTicketUsed(ticket.id)
  return {
    status: 'valid',
    title: 'Valid',
    message: `${ticket.id} · ${ticket.ticketType} · admitted, attendance recorded.`,
    ticket: admitted ?? ticket,
  }
}

import { events as SEED_EVENTS } from '../mocks/events'
import type { Event, TicketType } from '../types/event'

export type EventStatus = 'Draft' | 'Pending' | 'Published'

export interface StoredEvent extends Event {
  status: EventStatus
}

export interface EventInput {
  id?: string
  title: string
  description: string
  category: string
  city: string
  venue: string
  date: string
  time: string
  organizer: string
  image: string
  status: EventStatus
  ticketTypes: TicketType[]
}

export const EVENTS_KEY = 'tm.ticket-master.events'

const seed = (): StoredEvent[] => SEED_EVENTS.map((event) => ({ ...event, status: 'Published' }))

function hasStorage(): boolean {
  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined'
}

function readEvents(): StoredEvent[] {
  if (!hasStorage()) return seed()
  const raw = window.localStorage.getItem(EVENTS_KEY)
  if (!raw) {
    const initial = seed()
    window.localStorage.setItem(EVENTS_KEY, JSON.stringify(initial))
    return initial
  }
  try {
    const parsed = JSON.parse(raw) as StoredEvent[]
    return Array.isArray(parsed) && parsed.length > 0 ? parsed : seed()
  } catch {
    const initial = seed()
    window.localStorage.setItem(EVENTS_KEY, JSON.stringify(initial))
    return initial
  }
}

function writeEvents(items: StoredEvent[]): void {
  if (!hasStorage()) return
  window.localStorage.setItem(EVENTS_KEY, JSON.stringify(items))
}

export function listAllEvents(): StoredEvent[] {
  return readEvents()
}

export function listEvents(): StoredEvent[] {
  return readEvents().filter((event) => event.status === 'Published')
}

export function getEvent(id: string): StoredEvent | undefined {
  return readEvents().find((event) => event.id === id)
}

function makeId(title: string): string {
  const slug = title.toLowerCase().trim().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')
  return slug || `event-${Date.now()}`
}

export function saveEvent(input: EventInput): StoredEvent {
  const events = readEvents()
  const existing = input.id ? events.find((event) => event.id === input.id) : undefined
  const saved: StoredEvent = {
    id: existing?.id ?? input.id ?? makeId(input.title),
    title: input.title.trim(),
    description: input.description,
    category: input.category,
    city: input.city,
    venue: input.venue,
    date: input.date,
    time: input.time,
    organizer: input.organizer,
    image: input.image || '/images/events/event-1.webp',
    status: input.status,
    ticketTypes: input.ticketTypes.map((type) => ({ ...type })),
  }
  if (existing) {
    writeEvents(events.map((event) => (event.id === existing.id ? saved : event)))
  } else {
    writeEvents([saved, ...events])
  }
  return saved
}

export function deleteEvent(id: string): void {
  writeEvents(readEvents().filter((event) => event.id !== id))
}

export function incrementSold(eventId: string, ticketTypeId: string, quantity: number): void {
  const events = readEvents()
  const event = events.find((candidate) => candidate.id === eventId)
  if (!event) return
  event.ticketTypes = event.ticketTypes.map((type) =>
    type.id === ticketTypeId ? { ...type, sold: Math.min(type.capacity, type.sold + quantity) } : type,
  )
  writeEvents([...events])
}

export function resetEvents(): void {
  writeEvents(seed())
}

function hasStorage(): boolean {
  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined'
}

function readList<T>(key: string, seed: T[]): T[] {
  if (!hasStorage()) return [...seed]

  const raw = window.localStorage.getItem(key)
  if (!raw) {
    window.localStorage.setItem(key, JSON.stringify(seed))
    return [...seed]
  }

  try {
    const parsed = JSON.parse(raw) as T[]
    return Array.isArray(parsed) ? parsed : [...seed]
  } catch {
    window.localStorage.setItem(key, JSON.stringify(seed))
    return [...seed]
  }
}

function writeList<T>(key: string, items: T[]): void {
  if (!hasStorage()) return
  window.localStorage.setItem(key, JSON.stringify(items))
}

function generateId(prefix: string): string {
  const serial = String(Math.floor(100000 + Math.random() * 900000))
  return `${prefix}-2026-${serial}`
}

export interface IssueTicketInput {
  eventId: string
  eventName: string
  date: string
  time: string
  venue: string
  city: string
  ticketType: string
  price: number
  holder: string
}

export function issueTicket(input: IssueTicketInput): TicketRecord {
  const tickets = listTickets()
  let id = generateId('TM')
  while (tickets.some((ticket) => ticket.id === id)) {
    id = generateId('TM')
  }

  const ticket: TicketRecord = {
    ...input,
    id,
    status: 'valid',
  }

  writeList(TICKETS_KEY, [ticket, ...tickets])
  return ticket
}

export function markTicketUsed(id: string): TicketRecord | undefined {
  const normalized = normalizeTicketCode(id)
  const tickets = listTickets()
  const index = tickets.findIndex((ticket) => ticket.id.toUpperCase() === normalized)
  if (index === -1) return undefined

  const updated: TicketRecord = {
    ...tickets[index],
    status: 'used',
    checkedInAt: new Date().toISOString(),
  }
  tickets[index] = updated
  writeList(TICKETS_KEY, tickets)
  return updated
}

export interface IssueBookingInput {
  event: Event
  ticketType: TicketType
  quantity: number
  holder: string
}

export interface IssueBookingResult {
  booking: BookingRecord
  tickets: TicketRecord[]
}

// Issues the tickets and the booking that owns them in one step.
export function issueBooking(input: IssueBookingInput): IssueBookingResult {
  const { event, ticketType, quantity, holder } = input

  const tickets: TicketRecord[] = []
  for (let index = 0; index < quantity; index += 1) {
    tickets.push(
      issueTicket({
        eventId: event.id,
        eventName: event.title,
        date: event.date,
        time: event.time,
        venue: event.venue,
        city: event.city,
        ticketType: ticketType.name,
        price: ticketType.price,
        holder: holder.trim() || 'Guest',
      }),
    )
  }

  const bookings = listBookings()
  let bookingId = generateId('BK')
  while (bookings.some((booking) => booking.id === bookingId)) {
    bookingId = generateId('BK')
  }

  const booking: BookingRecord = {
    id: bookingId,
    eventTitle: event.title,
    venue: event.venue,
    date: event.date,
    quantity,
    total: quantity * ticketType.price,
    status: 'confirmed',
    ticketIds: tickets.map((ticket) => ticket.id),
  }

  writeList(BOOKINGS_KEY, [booking, ...bookings])
  return { booking, tickets }
}

export type KYCStatus = 'Unverified' | 'Pending' | 'Verified' | 'Rejected';

export interface OrganizerKYC {
  userId: string;
  businessName: string;
  phoneNumber: string;
  nationalId: string;
  status: KYCStatus;
  submittedAt?: string;
}

const KYC_KEY = 'tm.ticket-master.kyc';

function hasStorage(): boolean {
  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined';
}

export function getKYC(userId: string): OrganizerKYC | null {
  if (!hasStorage()) return null;
  const raw = window.localStorage.getItem(KYC_KEY);
  if (!raw) return null;
  try {
    const parsed = JSON.parse(raw) as OrganizerKYC[];
    return parsed.find((k) => k.userId === userId) ?? null;
  } catch {
    return null;
  }
}

export function saveKYC(data: OrganizerKYC): void {
  if (!hasStorage()) return;
  const raw = window.localStorage.getItem(KYC_KEY);
  let parsed: OrganizerKYC[] = [];
  if (raw) {
    try { parsed = JSON.parse(raw) as OrganizerKYC[]; } catch {}
  }
  const existingIndex = parsed.findIndex((k) => k.userId === data.userId);
  if (existingIndex >= 0) {
    parsed[existingIndex] = data;
  } else {
    parsed.push(data);
  }
  window.localStorage.setItem(KYC_KEY, JSON.stringify(parsed));
}

export function listAllKYC(): OrganizerKYC[] {
  if (!hasStorage()) return [];
  const raw = window.localStorage.getItem(KYC_KEY);
  if (!raw) return [];
  try { return JSON.parse(raw) as OrganizerKYC[]; } catch { return []; }
}

import { createRootRoute, createRoute, createRouter } from '@tanstack/react-router';
import { AppLayout } from '../components/AppLayout';
import { LoginPage, requireAuth } from '../features/authentication';
import { EventsPage, EventDetailsPage } from '../features/events';
import { CheckoutPage } from '../features/checkout';
import { MyTicketsPage, BookingHistoryPage } from '../features/tickets';
import { ScannerPage } from '../features/validation';
import { OrganizerDashboardPage, CreateEventPage } from '../features/organizer';
import { HomePage } from '../features/landing'
import { AdminDashboardPage } from '../features/admin/pages/AdminDashboardPage';
import NotFoundPage from '../components/NotFoundPage';
import type { ReactNode } from 'react';

const rootRoute = createRootRoute({ component: AppLayout });

const page = <P extends string>(path: P, component: () => ReactNode, beforeLoad?: ReturnType<typeof requireAuth>) => createRoute({ getParentRoute: () => rootRoute, path, component, beforeLoad });

const indexRoute = createRoute({ getParentRoute: () => rootRoute, path: '/', component: HomePage });
const loginRoute = createRoute({
  getParentRoute: () => rootRoute,
  path: '/login',
  component: LoginPage,
  validateSearch: (search: Record<string, unknown>): { redirect?: string } => ({ redirect: typeof search.redirect === 'string' ? search.redirect : undefined }),
});
const notFoundRoute = createRoute({ getParentRoute: () => rootRoute, path: '*', component: NotFoundPage })

const routeTree = rootRoute.addChildren([
  indexRoute,
  loginRoute,
  page('/events', EventsPage),
  page('/events/$eventId', EventDetailsPage),
  page('/checkout/$eventId', CheckoutPage, requireAuth('customer')),
  page('/tickets', MyTicketsPage, requireAuth('customer')),         // <-- Added base tickets route
  page('/tickets/$ticketId', MyTicketsPage, requireAuth('customer')), // <-- Kept specific route
  page('/bookings', BookingHistoryPage, requireAuth('customer')),
  page('/organizer/dashboard', OrganizerDashboardPage, requireAuth('organizer')),
  page('/organizer/events/new', CreateEventPage, requireAuth('organizer')),
  page('/organizer/events/$eventId/edit', CreateEventPage, requireAuth('organizer')),
  page('/admin', AdminDashboardPage, requireAuth('admin')),
  page('/scan', ScannerPage, requireAuth('staff')),
  notFoundRoute,
]);

export const router = createRouter({ routeTree });

declare module '@tanstack/react-router' {
  interface Register { router: typeof router; }
}

=============== src/types/event.ts ===============
export interface TicketType {
  id: string
  name: string
  price: number
  capacity: number
  sold: number
}

export interface Event {
  id: string
  title: string
  category: string
  city: string
  venue: string
  date: string // ISO date, e.g. 2026-11-14
  time: string // 24h, e.g. 18:00
  description: string
  organizer: string
  image: string
  ticketTypes: TicketType[]
}

// cheapest ticket type price, used for cards, filters and sorting
export function fromPrice(event: Event): number {
  if (event.ticketTypes.length === 0) return 0
  return Math.min(...event.ticketTypes.map((type) => type.price))
}

export function remainingSeats(type: TicketType): number {
  return Math.max(0, type.capacity - type.sold)
}

export function isSoldOut(event: Event): boolean {
  return (
    event.ticketTypes.length > 0 &&
    event.ticketTypes.every((type) => remainingSeats(type) === 0)
  )
}

// true when the event date is today or later
export function isUpcomingEvent(event: Event): boolean {
  const today = new Date().toLocaleDateString('en-CA') // YYYY-MM-DD
  return event.date >= today
}
